/* Cached, streamed WebGL world with real shadow mapping and curvature. */
window.createTempleRenderer=function(canvas){
const gl=canvas.getContext('webgl',{antialias:true,alpha:false,powerPreference:'high-performance'});if(!gl)return null;
const vs=`attribute vec3 p;attribute vec3 n;attribute vec3 c;uniform mat4 vp;uniform mat4 lightVP;uniform float offsetZ;varying vec3 N;varying vec3 C;varying vec3 P;varying vec4 L;void main(){P=p+vec3(0.,0.,offsetZ);float bend=max(P.z-32.,0.);P.y-=bend*bend*.0028;N=normalize(vec3(n.x,n.y,n.z+2.*.0028*bend*n.y));C=c;L=lightVP*vec4(P,1.);gl_Position=vp*vec4(P,1.);}`;
const fs=`precision highp float;varying vec3 N;varying vec3 C;varying vec3 P;varying vec4 L;uniform vec3 sky;uniform vec3 eye;uniform sampler2D shadowMap;uniform float shadowEnabled;uniform float texel;uniform float night;
float unpackDepth(vec4 v){return dot(v,vec4(1.,1./255.,1./65025.,1./16581375.));}
void main(){vec3 normal=normalize(N);vec3 light=normalize(vec3(-.6,1.,-.5));float diffuse=max(dot(normal,light),0.);float shade=1.;vec3 q=L.xyz/L.w*.5+.5;if(shadowEnabled>.5&&q.x>.002&&q.x<.998&&q.y>.002&&q.y<.998&&q.z>0.&&q.z<1.){float bias=max(.00065,.002*(1.-diffuse));shade=0.;for(int x=-1;x<=1;x++){for(int y=-1;y<=1;y++){float dep=unpackDepth(texture2D(shadowMap,q.xy+vec2(float(x),float(y))*texel));shade+=q.z-bias<=dep?1.:.28;}}shade/=9.;}float rim=pow(1.-max(dot(normal,normalize(eye-P)),0.),3.);vec3 color=C*(.35+diffuse*.72*shade)+vec3(1.,.83,.58)*rim*.065;float shine=pow(max(dot(reflect(-light,normal),normalize(eye-P)),0.),32.);if(C.r>C.b*1.4&&C.g>C.b*1.1)color+=vec3(1.,.85,.55)*shine*.25*shade;float fog=smoothstep(48.,115.,P.z);gl_FragColor=vec4(mix(color,sky,fog),1.);}`;
const depthFS=`precision highp float;void main(){vec4 enc=fract(gl_FragCoord.z*vec4(1.,255.,65025.,16581375.));enc-=enc.yzww*vec4(1./255.,1./255.,1./255.,0.);gl_FragColor=enc;}`;
function shader(type,source){const s=gl.createShader(type);gl.shaderSource(s,source);gl.compileShader(s);if(!gl.getShaderParameter(s,gl.COMPILE_STATUS))throw Error(gl.getShaderInfoLog(s));return s;}
function makeProgram(fragment){const pr=gl.createProgram();gl.attachShader(pr,shader(gl.VERTEX_SHADER,vs));gl.attachShader(pr,shader(gl.FRAGMENT_SHADER,fragment));gl.linkProgram(pr);if(!gl.getProgramParameter(pr,gl.LINK_STATUS))throw Error('WebGL link failed');return {pr,attrs:['p','n','c'].map(n=>gl.getAttribLocation(pr,n)),uniforms:Object.fromEntries(['vp','lightVP','offsetZ','sky','eye','shadowMap','shadowEnabled','texel','night'].map(n=>[n,gl.getUniformLocation(pr,n)]))};}
const mainProgram=makeProgram(fs),depthProgram=makeProgram(depthFS);
gl.enable(gl.DEPTH_TEST);gl.disable(gl.CULL_FACE);
const shadowSize=canvas.clientWidth<600?512:1024,shadowTexture=gl.createTexture();gl.bindTexture(gl.TEXTURE_2D,shadowTexture);gl.texImage2D(gl.TEXTURE_2D,0,gl.RGBA,shadowSize,shadowSize,0,gl.RGBA,gl.UNSIGNED_BYTE,null);gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MIN_FILTER,gl.NEAREST);gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MAG_FILTER,gl.NEAREST);gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_S,gl.CLAMP_TO_EDGE);gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_T,gl.CLAMP_TO_EDGE);
const shadowDepth=gl.createRenderbuffer();gl.bindRenderbuffer(gl.RENDERBUFFER,shadowDepth);gl.renderbufferStorage(gl.RENDERBUFFER,gl.DEPTH_COMPONENT16,shadowSize,shadowSize);const shadowFB=gl.createFramebuffer();gl.bindFramebuffer(gl.FRAMEBUFFER,shadowFB);gl.framebufferTexture2D(gl.FRAMEBUFFER,gl.COLOR_ATTACHMENT0,gl.TEXTURE_2D,shadowTexture,0);gl.framebufferRenderbuffer(gl.FRAMEBUFFER,gl.DEPTH_ATTACHMENT,gl.RENDERBUFFER,shadowDepth);const hasShadows=gl.checkFramebufferStatus(gl.FRAMEBUFFER)===gl.FRAMEBUFFER_COMPLETE;gl.bindFramebuffer(gl.FRAMEBUFFER,null);
let vertices=[],clock=0,camX=0,camFov=52,lastFrame=0,worldMap=-1,frameCounter=0;const chunks=new Map();let backdrop=null,ground=null;
const stats={generatedChunks:0,dynamicVertices:0,cachedVertices:0,drawCalls:0,chunkOffsets:[],shadowMap:hasShadows};
const gold=[.88,.59,.16],goldLight=[1,.8,.36],red=[.57,.055,.07],wood=[.23,.095,.045],skin=[.74,.47,.29],ink=[.065,.09,.09],white=[.88,.86,.75];
function mesh(){const buffer=gl.createBuffer(),data=new Float32Array(vertices);gl.bindBuffer(gl.ARRAY_BUFFER,buffer);gl.bufferData(gl.ARRAY_BUFFER,data,gl.STATIC_DRAW);return {buffer,count:data.length/9};}
const dynamic={buffer:gl.createBuffer(),count:0},effects={buffer:gl.createBuffer(),count:0};let dynamicData=new Float32Array(9*60000),effectData=new Float32Array(9*16000);gl.bindBuffer(gl.ARRAY_BUFFER,dynamic.buffer);gl.bufferData(gl.ARRAY_BUFFER,dynamicData.byteLength,gl.DYNAMIC_DRAW);gl.bindBuffer(gl.ARRAY_BUFFER,effects.buffer);gl.bufferData(gl.ARRAY_BUFFER,effectData.byteLength,gl.DYNAMIC_DRAW);
function upload(target,effect=false){let data=effect?effectData:dynamicData;if(vertices.length>data.length){data=new Float32Array(vertices.length*2);if(effect)effectData=data;else dynamicData=data;gl.bindBuffer(gl.ARRAY_BUFFER,target.buffer);gl.bufferData(gl.ARRAY_BUFFER,data.byteLength,gl.DYNAMIC_DRAW);}data.set(vertices);target.count=vertices.length/9;gl.bindBuffer(gl.ARRAY_BUFFER,target.buffer);gl.bufferSubData(gl.ARRAY_BUFFER,0,data.subarray(0,vertices.length));}
function submit(program,m,offset=0){if(!m||!m.count)return;gl.bindBuffer(gl.ARRAY_BUFFER,m.buffer);program.attrs.forEach((location,i)=>{if(location>=0){gl.enableVertexAttribArray(location);gl.vertexAttribPointer(location,3,gl.FLOAT,false,36,i*12);}});gl.uniform1f(program.uniforms.offsetZ,offset);gl.drawArrays(gl.TRIANGLES,0,m.count);stats.drawCalls++;}
